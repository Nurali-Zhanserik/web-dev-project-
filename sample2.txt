payment_build.sh

#!/bin/bash

set -euo pipefail

IMAGE_NAME='payment'
DATE_TAG=$(date +%Y.%m.%d)
REPO_DIR=$(mktemp -d)

echo "Remove old image"
podman image rm "${IMAGE_NAME}:latest" || true

echo "Start clonning..."
git clone https://github.com/dvalyayevkbtu/payment $REPO_DIR

cp ~/Dockerfile $REPO_DIR

echo "Start building..."
podman build -t "${IMAGE_NAME}:latest" -t "${IMAGE_NAME}:${DATE_TAG}" "${REPO_DIR}"

echo "Remove dangling images after multi-stage build"
podman image prune -f

rm -rf "$REPO_DIR"
echo "Done: ${IMAGE_NAME}:latest ${IMAGE_NAME}:${DATE_TAG}"
